# add id_rsa

# increase swap to RAM size, if less, use ol_sage, extend if needed
lvcreate -n lv_swap -L 16G ol_sage
mkswap /dev/ol_sage/lv_swap
swapoff /dev/ol/swap
swapon /dev/ol_sage/lv_swap
nano /etc/fstab
	change /ol/swap to /ol_sage/lv_swap
nano /etc/default/grub
	change /ol/swap references to ol_sage/lv_swap
grub2-mkconfig > /boot/grub2/grub.cfg
lvremove /dev/ol/swap
swapon -s

# assign the rest of disk to a pv
fdisk -l
fdisk /dev/sda
	n
	4,xx,xx
	t,4,lvm
	w
pvcreate /dev/sda4
pvs
vgcreate vg_oracle /dev/sda4
vgs
pvscan
lvcreate -n lv_oracle -L 60G vg_oracle
lvcreate -n lv_oracle_data -L 60G vg_oracle
lvcreate -n lv_oracle_fra -L 60G vg_oracle
lvcreate -n lv_oracle_backup -L 60G vg_oracle
lvs
lvdisplay
mkfs.xfs /dev/vg_oracle/lv_oracle
mkfs.xfs /dev/vg_oracle/lv_oracle_data
mkfs.xfs /dev/vg_oracle/lv_oracle_fra
mkfs.xfs /dev/vg_oracle/lv_oracle_backup

mkdir /uora
mkdir /uoradata
mkdir /uorafra
mkdir /uorabackup

# mount them in fstab
nano /etc/fstab
/dev/vg_oracle/lv_oracle	/uora	xfs	defaults	0 0
/dev/vg_oracle/lv_oracle_data	/uoradata	xfs	defaults	0 0
/dev/vg_oracle/lv_oracle_fra	/uorafra	xfs	defaults	0 0
/dev/vg_oracle/lv_oracle_backup	/uorabackup	xfs	defaults	0 0

# increase size of /dev/shm to RAM size, if less
nano /etc/fstab
none		/dev/shm		tmpfs	defaults,size=16G	0 0

# reboot or
systemctl daemon-reload
mount -o remount /dev/shm
df -h
