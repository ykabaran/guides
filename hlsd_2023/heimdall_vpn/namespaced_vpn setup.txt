nano /usr/local/sbin/setup-frankfurt-netns.sh
#!/bin/bash
set -e

NS="frankfurt"
HOST_IF="veth-fra-h"
NS_IF="veth-fra-n"

# Create namespace if missing
if ! ip netns list | grep -q "^${NS} "; then
    ip netns add "$NS"
fi

# Create veth pair if missing
if ! ip link show "$HOST_IF" >/dev/null 2>&1; then
    ip link add "$HOST_IF" type veth peer name "$NS_IF"
    ip link set "$NS_IF" netns "$NS"
fi

# Configure host side
ip addr replace 10.28.10.1/30 dev "$HOST_IF"
ip link set "$HOST_IF" up

# Configure namespace side
ip -n "$NS" addr replace 10.28.10.2/30 dev "$NS_IF"
ip -n "$NS" link set "$NS_IF" up
ip -n "$NS" link set lo up

# Configure namespace gateway
ip -n "$NS" route replace default via 10.28.10.1

# Enable below line if you dont want fallback routes to active in case of VPN disconnection
#ip -n "$NS" route replace blackhole 159.223.234.216/32 metric 100

echo "Frankfurt namespace configured"

chmod 755 /usr/local/sbin/setup-frankfurt-netns.sh

/usr/local/sbin/setup-frankfurt-netns.sh

nano /etc/systemd/system/frankfurt-netns.service
[Unit]
Description=Frankfurt Network Namespace
Wants=network-online.target
After=network-online.target
Before=frankfurt-openvpn.service

[Service]
Type=oneshot
ExecStart=/usr/local/sbin/setup-frankfurt-netns.sh
RemainAfterExit=yes

[Install]
WantedBy=multi-user.target

systemctl daemon-reload
systemctl enable --now frankfurt-netns.service

# enable forwarding with ufw
ufw route allow \
  in on veth-fra-h \
  out on ens224 \
  from 10.28.10.0/30 \
  to any

ufw reload

[Unit]
Description=OpenVPN Client inside Frankfurt Namespace

After=frankfurt-netns.service network-online.target

[Service]

[Install]
