ufw enable

apt install openvpn

cd /root
mkdir pki
cd pki
mkdir ca server client
cd ca
mkdir certs private csr newcerts issued
chmod 700 private
touch index.txt
echo 1000 > serial
echo 1000 > crlnumber
nano openssl.conf
# copy ca-openssl.conf file here
openssl genrsa -aes256 -out private/ca.key 4096
chmod 400 private/ca.key
openssl req -config openssl.conf -key private/ca.key -new -x509 -days 7300 -sha256 -extensions v3_ca -out certs/ca.crt

# verify
openssl x509 -noout -text -in certs/ca.crt

cd ../server
nano openssl.conf
# copy vpn-server-openssl.conf file here

openssl genrsa -out server.key 4096
openssl req -config openssl.conf -new -sha256 -key server.key -out server.csr
cp server.csr ../ca/csr/server.vpn.heimdall.hlsd.com.csr
cd ../ca
openssl ca -config openssl.conf -extensions server_cert -days 3750 -notext -md sha256 -in csr/server.vpn.heimdall.hlsd.com.csr -out issued/server.vpn.heimdall.hlsd.com.crt
openssl ca -config openssl.conf -gencrl -out crl.pem
cp certs/ca.crt crl.pem ../server/
cp issued/server.vpn.heimdall.hlsd.com.crt ../server/server.crt

cd /etc/openvpn
cp /root/pki/server/server.key /root/pki/server/ca.crt /root/pki/server/crl.pem /root/pki/server/server.crt ./
mkdir ccd

openvpn --genkey secret ta.key

nano server.conf
# copy vpn-server-openvpn.conf file here
nano /usr/share/doc/openvpn/examples/sample-config-files/server.conf
# compare to see if any options look wrong

  in on tun0 \
  out on eth0 \

chmod u+x vpn-make-client-ovpn.sh

# make client csrs
./vpn-make-client-csr.sh yildiz.heimdall.hlsd.com

# get csrs signed with ca
cp yildiz.vpn.heimdall.hlsd.com/client.csr ../ca/csr/yildiz.vpn.heimdall.hlsd.com.csr

cd ../ca
openssl ca -config openssl.conf -extensions client_cert -days 3750 -notext -md sha256 -in csr/yildiz.vpn.heimdall.hlsd.com.csr -out issued/yildiz.vpn.heimdall.hlsd.com.crt

# revoke one just in case
openssl ca -config openssl.conf -revoke issued/revoked.vpn.heimdall.hlsd.com.crt
mv issued/revoked.vpn.heimdall.hlsd.com.cert.pem revoked/20261104_revoked.vpn.heimdall.hlsd.com.crt

openssl ca -config openssl.conf -gencrl -out crl.pem
cp crl.pem ../server/
cp crl.pem /root/pki/server/server.crt /etc/openvpn/

# copy the signed certs back together with crl
cp issued/yildiz.vpn.heimdall.hlsd.com.crt ../client/yildiz.vpn.heimdall.hlsd.com/client.crt

# make client ovpns
